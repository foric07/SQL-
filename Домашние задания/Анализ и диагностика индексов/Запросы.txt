ЗАПРОС: 
EXPLAIN analyze SELECT
    c.first_name, 
    c.last_name, 
    m.title,
    COUNT(r.rental_id) AS rental_count, 
    MAX(r.rental_date) AS last_rental_date
FROM 
    "KINO".customers c
JOIN 
    "KINO".rentals r ON c.customer_id = r.customer_id
JOIN 
    "KINO".movies m ON r.movie_id = m.movie_id
WHERE 
    m.genre = 'Action'
    AND r.rental_date BETWEEN '2021-01-01' AND '2022-12-31'
GROUP BY 
    c.first_name,
    c.last_name, 
    m.title
ORDER BY 
    rental_count DESC, 
    last_rental_date DESC
LIMIT 10;
