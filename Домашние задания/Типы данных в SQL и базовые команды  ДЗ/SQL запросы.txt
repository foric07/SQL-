-- 1. Все актёры
SELECT * 
FROM Actor;

-- 2. Фильмы жанра "Драма" после 2010 года
SELECT f.*
FROM Film f
JOIN Film_Genre fg ON f.id = fg.film_id
JOIN Genre g ON g.id = fg.genre_id
WHERE g.name = 'Драма' AND f.release_year > 2010;

-- 3. Актёры, отсортированные по фамилии
SELECT * 
FROM Actor 
ORDER BY last_name;

-- 4. Топ-5 фильмов по рейтингу
SELECT * 
FROM Film 
ORDER BY rating DESC 
LIMIT 5;

-- 5. Топ фильмов по рейтингу (с 6 по 10)
SELECT * 
FROM Film 
ORDER BY rating DESC 
OFFSET 5 
LIMIT 5;
