-- Таблица фильмов
CREATE TABLE Film (
    id SERIAL PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    release_year INT NOT NULL,
    rating DECIMAL(3,1),
    duration INT NOT NULL,
    description TEXT
);

-- Таблица жанров
CREATE TABLE Genre (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    description TEXT
);

-- Промежуточная таблица "фильм-жанр"
CREATE TABLE Film_Genre (
    id SERIAL PRIMARY KEY,
    film_id INT NOT NULL REFERENCES Film(id) ON DELETE CASCADE,
    genre_id INT NOT NULL REFERENCES Genre(id) ON DELETE CASCADE
);

-- Таблица актёров
CREATE TABLE Actor (
    id SERIAL PRIMARY KEY,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    birth_date DATE,
    nationality VARCHAR(100)
);

-- Промежуточная таблица "фильм-актёр"
CREATE TABLE Film_Actor (
    id SERIAL PRIMARY KEY,
    film_id INT NOT NULL REFERENCES Film(id) ON DELETE CASCADE,
    actor_id INT NOT NULL REFERENCES Actor(id) ON DELETE CASCADE
);

-- Таблица режиссёров
CREATE TABLE Director (
    id SERIAL PRIMARY KEY,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    birth_date DATE,
    nationality VARCHAR(100)
);

-- Промежуточная таблица "фильм-режиссёр"
CREATE TABLE Film_Director (
    id SERIAL PRIMARY KEY,
    film_id INT NOT NULL REFERENCES Film(id) ON DELETE CASCADE,
    director_id INT NOT NULL REFERENCES Director(id) ON DELETE CASCADE
);

-- Таблица клиентов
CREATE TABLE Customer (
    id SERIAL PRIMARY KEY,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    email VARCHAR(255) NOT NULL UNIQUE,
    phone_number VARCHAR(20),
    address TEXT
);

-- Таблица заказов (прокатов)
CREATE TABLE Rental (
    id SERIAL PRIMARY KEY,
    customer_id INT NOT NULL REFERENCES Customer(id),
    film_id INT NOT NULL REFERENCES Film(id),
    rental_date DATE NOT NULL,
    return_date DATE
);
